Maximize daily profit.

from pulp import *

# Create the LP problem
prob = LpProblem("Furniture Workshop", LpMaximize)

# Define decision variables
chair_var = LpVariable('Chair', lowBound=0, cat='Integer')
table_var = LpVariable('Table', lowBound=0, cat='Integer')

# Set the objective function
prob += 20 * chair_var + 30 * table_var, "Total Profit"

# Add constraints
prob += 2 * chair_var + 4 * table_var <= 24, "Labor Constraint"
prob += 3 * chair_var + 5 * table_var <= 30, "Wood Constraint"

# Solve the problem
prob.solve(pulp.PULP_CBC_CMD(msg=0))

# Store the result
_result = {
    "status": pulp.LpStatus[prob.status],
    "objective": pulp.value(prob.objective),
    "variables": {v.name: pulp.value(v) for v in prob.variables()}
}
