prob = pulp.LpProblem("furniture_factory", pulp.LpMaximize)

products = [
    "Dining_Chair", "Dining_Table", "Office_Desk", "Bookcase", "Wardrobe",
    "Armchair", "Bed_Frame", "Coffee_Table", "Storage_Cabinet", "Lounge_Sofa"
