from pulp import *

# Data
supply = {
    1: 100,
    2: 80,
    3: 90,
    4: 70,
    5: 60,
    6: 85,
    7: 75,
    8: 65
}

demand = {
    1: 120,
    2: 110,
    3: 130,
    4: 140,
    5: 125
}

cost = {
    1: {1: 4, 2: 2, 3: 6, 4: 9, 5: 5},
    2: {1: 7, 2: 5, 3: 3, 4: 6, 5: 8},
    3: {1: 3, 2: 8, 3: 2, 4: 7, 5: 4},
    4: {1: 9, 2: 4, 3: 7, 4: 2, 5: 10},
    5: {1: 5, 2: 3, 3: 8, 4: 4, 5: 6},
    6: {1: 6, 2: 7, 3: 4, 4: 5, 5: 2},
    7: {1: 8, 2: 6, 3: 5, 4: 3, 5: 7},
    8: {1: 2, 2: 9, 3: 7, 4: 8, 5: 3}
}

# Create the LP problem
prob = LpProblem("Transportation Problem", pulp.LpMinimize)

# Decision variables
x = {}
for i in supply:
    for j in demand:
        x[i, j] = LpVariable(f"x_{i}_{j}", lowBound=0)

# Objective function
prob += pulp.lpSum([cost[i][j] * x[i, j] for i in supply for j in demand])

# Supply constraints
for i in supply:
    prob += pulp.lpSum([x[i, j] for j in demand]) <= supply[i]

# Demand constraints
for j in demand:
    prob += pulp.lpSum([x[i, j] for i in supply]) == demand[j]

# Solve the problem
prob.solve(pulp.PULP_CBC_CMD(msg=0))

# Store the results
_result = {
    "status": pulp.LpStatus[prob.status],
    "objective": pulp.value(prob.objective),
    "variables": {v.name: pulp.value(v) for v in prob.variables()}
}
