**Problem:** A company has two warehouses and three retail stores. Warehouse W1 has 80 units of inventory; Warehouse W2 has 60 units. Minimize total shipping cost.

**Shipping plan (units):**

import pulp

# Create the problem
prob = pulp.LpProblem("Warehouse_Routing", pulp.LpMinimize)

# Define decision variables
# x[i][j] = units shipped from source i to destination j
x = {}
for i in [1, 2]: # Warehouses W1 and W2
    for j in [1, 2, 3]: # Stores S1, S2, and S3
        x[i, j] = pulp.LpVariable(f"x_{i}_{j}", lowBound=0)

# Define the objective function
# Minimize total shipping cost
cost = {
    (1, 1): 2,  # W1 to S1
    (1, 2): 3,  # W1 to S2
    (1, 3): 1,  # W1 to S3
    (2, 1): 5,  # W2 to S1
    (2, 2): 4,  # W2 to S2
    (2, 3): 8   # W2 to S3
}
objective = pulp.lpSum([cost[i, j] * x[i, j] for i in [1, 2] for j in [1, 2, 3]])
prob += objective

# Define the supply constraints
# For each source (warehouse), the total shipped must be <= supply
supply = {
    1: 80,  # W1
    2: 60   # W2
}
for i in [1, 2]:
    prob += pulp.lpSum([x[i, j] for j in [1, 2, 3]]) <= supply[i]

# Define the demand constraints
# For each destination (store), the total received must == demand
demand = {
    1: 50,  # S1
    2: 40,  # S2
    3: 50   # S3
}
for j in [1, 2, 3]:
    prob += pulp.lpSum([x[i, j] for i in [1, 2]]) == demand[j]

# Solve the problem
prob.solve()

# Store the result
_result = {
    "status": pulp.LpStatus[prob.status],
    "objective": pulp.value(prob.objective),
    "variables": {v.name: pulp.value(v) for v in prob.variables()}
}
